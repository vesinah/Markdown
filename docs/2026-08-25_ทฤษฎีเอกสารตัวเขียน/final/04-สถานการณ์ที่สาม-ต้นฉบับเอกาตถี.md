# บทที่ 4: สถานการณ์ที่สาม — ต้นฉบับที่เหลืออยู่เพียงฉบับเดียว (Codex Unicus)

บทที่ผ่านมาได้อภิปรายสถานการณ์ทางต้นฉบับสองรูปแบบซึ่งระเบียบวิธีการชำระข้อความถูกออกแบบมารองรับโดยเฉพาะ ได้แก่ สถานการณ์ที่ข้อความหนึ่งเดียวสืบทอดมาถึงเราผ่านพยานตัวเขียนหลายฉบับซึ่งนำมาสอบเทียบกันได้ และสถานการณ์ที่ข้อความต้นฉบับสูญหายไปแต่ยังปรากฏผ่านฉบับแปลซึ่งทำหน้าที่เป็นพยานทุติยภูมิ บทนี้จะเข้าสู่สถานการณ์ที่สามอันเป็นกรณีขอบเขตที่ท้าทายสมมติฐานฐานรากของวิชาการชำระข้อความอย่างรุนแรงที่สุด นั่นคือเมื่อข้อความหนึ่งเหลือพยานตัวเขียนอยู่เพียงฉบับเดียวในโลก ในสถานการณ์เช่นนี้กระบวนการสอบเทียบฉบับซึ่งเป็นหัวใจของระเบียบวิธีลัคมานเนียนไม่อาจดำเนินการได้เลย เพราะการสอบเทียบต้องอาศัยพยานอย่างน้อยสองฝ่ายจึงจะแยกแยะการอ่านดั้งเดิมออกจากความคลาดเคลื่อนของผู้คัดลอกได้ เมื่อพยานมีเพียงฉบับเดียว นักชำระข้อความทุกคนจึงถูกบังคับให้ตอบคำถามชุดใหม่ทั้งหมด ได้แก่ ควรพิมพ์ข้อความตามที่พยานเพียงฉบับเดียวถือไว้โดยไม่แตะต้องหรือไม่ ควรแก้ไขจุดที่เชื่อว่าเสียหายด้วยการสันนิษฐานได้ถึงขอบเขตใด และควรพึ่งพาเครื่องมือทางวิทยาศาสตร์เพื่อฟื้นฟูตัวอักษรที่จางหายไปได้มากน้อยเพียงใด บทนี้จะวิเคราะห์ทั้งด้านทฤษฎีการชำระข้อความและกรณีศึกษาหลัก ตั้งแต่มหากาพย์แองโกล-แซกซันที่ผ่านเพลิงไหม้ บทกวีมหากาพย์ฝรั่งเศสที่เหลือสำนวนเดียวแท้จริง คลังบทกวีไอซ์แลนด์ที่มีช่องว่างใบขาด เอกสารที่ถูกไฟกลืนจนเหลือเพียงสำเนาแทน ไปจนถึงปมความจริงแท้ของศิลาจารึกเอกาตถีที่สำคัญที่สุดหลักหนึ่งของไทย เพื่อสร้างกรอบวิธีวิทยาที่สมบูรณ์สำหรับการรับมือกับพยานผู้เดียวดาย

## 4.1 นิยามและผลต่อระเบียบวิธี

ศัพท์เทคนิคที่ใช้เรียกต้นฉบับที่เหลืออยู่เพียงฉบับเดียวในวิชาการชำระข้อความสากลคือ **codex unicus** หรือในภาษาไทยอาจเรียกว่า **ต้นฉบับเอกาตถี** หมายถึงตัวเขียนที่เป็นพยานเดียวที่หลงเหลืออยู่ของข้อความใดข้อความหนึ่ง โดยไม่มีตัวเขียนอื่นที่สืบมาจากสายสืบเดียวกันเพื่อนำมาเทียบเคียง ความเป็นเอกาตถีมีหลายระดับความเข้มข้น ในระดับสูงสุดคือกรณีที่ข้อความทั้งหมดมีอยู่เฉพาะในตัวเขียนเล่มเดียวเท่านั้น เช่นมหากาพย์ *Beowulf* หรือบทกวีชุดของกวีเพิร์ล ในระดับรองคือกรณีที่ข้อความส่วนใหญ่อยู่ในพยานเดียวแต่มีพยานบางส่วนปรากฏที่อื่นเป็นการเฉพาะจุด เช่นบทกวีไอซ์แลนด์บางบทที่มีฉบับขนานใน *Hauksbók* และในระดับที่ซับซ้อนที่สุดคือกรณี **เอกาตถีโดยตัวแทน** ซึ่งพยานทางตรงสูญหายไปแล้วแต่สำเนาที่สืบมาจากพยานนั้นยังอยู่ครบสาย เช่นประเพณีตัวเขียนของกาตุลลุสที่จะได้อภิปรายต่อไป สิ่งที่ทุกระดับมีร่วมกันคือการขาดพยานคู่เทียบ อันเป็นเครื่องมือพื้นฐานที่วิธีการสอบเทียบฉบับต้องอาศัย

ผลต่อระเบียบวิธีของสถานการณ์นี้รุนแรงและทันที เมื่อไม่มีพยานที่สอง **วิธีสร้างแผนภาพสายสืบฉบับ** (*stemma codicum*) ซึ่งเป็นเครื่องมือมาตรฐานสำหรับจัดลำดับความสัมพันธ์ระหว่างตัวเขียนและสืบสร้างข้อความต้นฉบับโดยอาศัยข้อผิดพลาดร่วมที่ถ่ายทอดซ้ำกัน ย่อมไม่มีทางปฏิบัติการได้ เพราะแผนภาพสายสืบต้องการเส้นทางการถ่ายถอดหลายเส้นจึงจะระบุจุดตัดและทิศทางการคัดลอกได้ ด้วยพยานฉบับเดียว ทุกการอ่านที่ปรากฏบนหน้ากระดาษจึงมีสถานะเท่าเทียมกันในเชิงหลักฐาน ไม่ว่าจะถูกต้องมาแต่เดิมหรือเป็นความผิดพลาดของผู้คัดลอก นักชำระข้อความจึงต้องเปลี่ยนนิยามของงานจากการสืบสร้างข้อความที่สมมติว่าอยู่เบื้องหลังพยานหลายฉบับ มาเป็นการบันทึกและอธิบายข้อความที่พยานฉบับเดียวถือไว้จริง ๆ ซึ่งเรียกว่า **การชำระแบบทำสำเนาเทียบต้น** (*diplomatic editing*) หรือ **การชำระแบบยึดฉบับฐาน** (*copy-text editing*) กล่าวคือพิมพ์ข้อความตามที่พยานเอกาตถีเขียนไว้อย่างละเอียดถี่ถ้วนทุกตัวอักษร ทุกการลบเขียนทับ ทุกรอยเว้นช่องว่าง แล้วแยกส่วนที่เป็นการแก้ไขโดยผู้ชำระออกไปไว้ในเชิงอรรถหรือภาคผนวกอย่างชัดเจน แนวทางนี้มีต้นแบบประวัติศาสตร์ที่ชัดเจนในฉบับพิมพ์มหากาพย์โรแลนด์ของโฌแซ็ฟ เบดีเย ค.ศ. 1922 ซึ่งพิมพ์ข้อความจากตัวเขียนอ๊อกซฟอร์ดตามตัวอักษรโดยไม่ปรับปรุงตามจินตนาการของผู้ชำระ พร้อมภาพถ่ายพิมพ์ลึกของตัวเขียนทุกหน้าประกอบไว้คู่กัน [1] และมีแบบแผนขนานกันในงานภาพถ่ายจำลองตัวเขียนบทกวีชุดเพิร์ลโดยอิสราเอล กอลแลนซ์ ในชุด Early English Text Society เมื่อ ค.ศ. 1923 ซึ่งชื่อเรื่องเองก็ระบุสถานะพยานเดียวอย่างเปิดเผยว่าเป็นการจำลองจากตัวเขียนเพียงฉบับเดียวที่มีอยู่ [2]

รูปแบบการชำระสมัยใหม่สำหรับข้อความเอกาตถีได้รับการพัฒนาต่อยอดในฉบับมาตรฐานเพื่อการศึกษา เช่นฉบับบทกวีทั้งสี่ของกวีเพิร์ลโดยมัลคอม แอนดรูว์และโรนัลด์ วอลดรอน ซึ่งคงโครงสร้างยึดฉบับพยานเดียวแต่เพิ่มระบบคำอธิบายและคำแปลสำหรับผู้เรียน [3] ลักษณะเฉพาะที่สำคัญที่สุดของภาคนิทานในฉบับเอกาตถีคือมันไม่สามารถทำหน้าที่คัดเลือกการอ่านจากพยานหลายฝ่ายได้อีกต่อไป เพราะไม่มีพยานอื่นให้คัดเลือก หน้าที่ของมันจึงเปลี่ยนไปเป็นการบันทึกสถานะกายภาพของตัวอักษรแต่ละตัวว่าอ่านได้ชัด อ่านได้แต่จาง หรืออ่านไม่ได้เลย พร้อมระบุว่าการอ่านที่ผู้ชำระเสนอแทนตัวอักษรสึกกร่อนนั้นมาจากวิธีใด ไม่ว่าจะเป็นการอ่านซ้ำด้วยแสงอัลตราไวโอเลต การสันนิษฐานตามบริบท หรือการเทียบสำนวนขนาน การเปลี่ยนหน้าที่นี้ปรากฏชัดที่สุดในคำอธิบายภารกิจของโครงการอิเล็กทรอนิกส์เบโอวุลฟ์ซึ่งระบุไว้ว่า

> นอกเหนือจากภาพความละเอียดสูงของตัวเขียน ภาพรังสีอัลตราไวโอเลตยังเผยการอ่านที่ถูกลบหรือจางหายไปกว่าร้อยแห่ง ตลอดจนการใช้สารเคมีอย่างแพร่หลายบนแผ่นที่เสียหายอย่างวิกฤต ด้วยการทำให้แสงส่องจากด้านหลังเป็นดิจิทัล ตัวอักษรกว่า 1,300 ตัวและบางส่วนของตัวอักษรซึ่งเดิมถูกซ่อนโดยงานอนุรักษ์ ได้รับการคืนสภาพขึ้นมาแล้ว [4]

คำประกาศนี้สรุปแก่นของบทได้ตรงที่สุด นั่นคือในสถานการณ์เอกาตถี ความก้าวหน้าของความรู้ทางข้อความไม่ได้เกิดจากการค้นพบพยานใหม่เสมอไป แต่เกิดจากการอ่านพยานเดิมซ้ำด้วยเครื่องมือที่ดีกว่าเดิม และการอ่านซ้ำนี้เองที่ทำหน้าที่แทนการสอบเทียบฉบับซึ่งไม่มีทางเป็นไปได้

## 4.2 กรณีศึกษาหลัก

### เบโอวุลฟ์: พยานเดียวที่ผ่านไฟ

มหากาพย์แองโกล-แซกซันเรื่อง *Beowulf* หลงเหลืออยู่ในสำเนายุคกลางเพียงฉบับเดียว คือตัวเขียนที่เรียกกันว่า **โนเวลล์โคเด็กซ์** ทะเบียน London, British Library, Cotton MS Vitellius A.xv แผ่น 94 หน้าด้านถึง 209 หน้าด้าน เป็นโคเด็กซ์สองเล่มที่มัดรวมกันในหอสมุดของซอยร์ รอเบิร์ต คอตตอน ตัวเขียนส่วนที่บรรจุ *Beowulf* ถูกคัดลอกในอังกฤษช่วงปลายศตวรรษที่ 10 ถึงไต้ศตวรรษที่ 11 โดยห้องสมุดบริติชกำหนดช่วงอายุไว้ 990–1024 และระบุว่ามีคนคัดลอกสองคนทำงานร่วมสมัยกัน คนแรกเขียนจากแผ่น 94 หน้าด้านถึงแผ่น 175 หน้าด้านบรรทัดที่ 3 และคนที่สองเขียนต่อจนถึงแผ่นสุดท้าย เนื้อหาในเล่มประกอบด้วยเทศนาเศษส่วนว่าด้วยนักบุญคริสโตเฟอร์ หนังสือสิ่งมหัศจรรย์แห่งตะวันออกพร้อมภาพเขียนเส้น จดหมายอเล็กซานเดอร์ถึงอริสโตเติล บท *Beowulf* ที่แผ่น 132 หน้าด้านถึง 201 หน้าด้าน และบท *Judith* ฉบับไม่สมบูรณ์ ร่องรอยความเป็นเจ้าของเริ่มจากชื่อและปี Laurence Nouell 1563 บนแผ่นแรก ก่อนเข้าสู่หอสมุดคอตตอนในตำแหน่งหีบรูปหล่อจักรพรรดิวิเทลลิอุส ชั้นเอ ตำแหน่งที่ 15 อันเป็นที่มาของเลขทะเบียน [5]

ประวัติความเสียหายของพยานฉบับนี้เป็นบทเรียนสำคัญที่สุดเรื่องหนึ่งของต้นฉบับศึกษา หอสมุดคอตตอนมอบให้แก่ชาติเมื่อ ค.ศ. 1700 แล้วถูกย้ายไปมาหลายครั้งจนถึงบ้านแอชเบิร์นฮัม ซึ่งเกิดเพลิงไหม้ใหญ่ในวันที่ 23 ตุลาคม ค.ศ. 1731 ไฟลวกขอบแผ่นตัวเขียนเสียหายหนัก ก่อนที่จอร์จ จอห์น ทอร์เคลินจะคัดลอกข้อความไว้สองชุดเมื่อ ค.ศ. 1787 ซึ่งกลายเป็นพยานสำคัญของการอ่านที่ยังอยู่ก่อนการสึกกร่อนเพิ่มขึ้น เมื่อถึงเวลาการบูรณะของพิพิธภัณฑ์บริติช ตัวอักษรบริเวณขอบจำนวนมากได้ร่วงหล่นหายไปแล้ว การผูกเล่มใหม่เมื่อ ค.ศ. 1845 ติดแผ่นเข้ากับกรอบกระดาษรายแผ่น ซึ่งช่วยชะลอการเสื่อมสภาพแต่กลับปิดบังตัวอักษรที่ขอบกรอบไปในตัว ตัวเขียนถูกพยายามกำหนดเลขแผ่นอย่างน้อยหกครั้งกว่าจะได้เลขทางการเมื่อ ค.ศ. 1884 และมีการพิมพ์ภาพถ่ายโคโลไทป์โดยจูลีอุส ซูปิตซาในชุด Early English Text Society เมื่อ ค.ศ. 1882 ซึ่งยังพิมพ์ซ้ำเป็นฉบับที่สองเมื่อ ค.ศ. 1959 เป็นเครื่องมือมาตรฐานของนักวิชาการก่อนยุคดิจิทัล [6] งานของเควิน เคียร์แนนในหนังสือ *Beowulf and the Beowulf Manuscript* เดินหน้าต่อด้วยการพิสูจน์จากหลักฐานโคดิโคโลยีว่าคนคัดลอกคนที่สองมีบทบาทเป็นผู้รวบรวมเล่มด้วย จึงเสนอวันเขียนช่วงต้นศตวรรษที่ 11 ใกล้ ค.ศ. 1010 ข้อเสนอที่เปลี่ยนการอภิปรายการลงวันที่ตัวเขียนจากการเทียบอักษรวิธีเพียงอย่างเดียว ให้คำนึงถึงโครงสร้างทางกายภาพของเล่มด้วย [7]

โครงการอิเล็กทรอนิกส์เบโอวุลฟ์เริ่มต้นเมื่อ ค.ศ. 1992 และพิมพ์ครั้งแรกโดยห้องสมุดบริติชในรูปแบบซีดีรอมเมื่อ ค.ศ. 1999 ปัจจุบันอยู่รุ่นที่ห้าซึ่งเปิดใช้ฟรีออนไลน์ ความสำเร็จเชิงวิธีวิทยาของโครงการคือการพิสูจน์ว่าตัวอักษรที่ถูกปิดบังโดยกรอบอนุรักษ์ปี 1845 สามารถอ่านได้คืนโดยไม่ต้องแกะกรอบ ด้วยเทคนิคส่องแสงจากด้านหลังผ่านใยแก้วนำแสงร่วมกับภาพรังสีอัลตราไวโอเลต ซึ่งรวมกันเผยการอ่านที่ถูกลบหรือจางกว่า 100 แห่ง คืนตัวอักษรที่ถูกซ่อนกว่า 1,300 ตัว และบันทึกความเสียหายจากสารเคมีที่คนยุคก่อนทาลงเพื่อพยายามทำให้ตัวอักษรเด่นขึ้นแต่กลับกัดกร่อนหมึก นอกจากนี้ระบบยังรวมภาพและข้อความของสำเนายุคแรกทั้งหมด รวมทั้งสำเนาทอร์เคลินสองชุด อันเปลี่ยนสำเนายุคศตวรรษที่ 18 ให้ทำหน้าที่เป็นพยานเสริมของตัวเขียนเดิมได้อย่างเป็นระบบ [8] ฉบับชำระมาตรฐานสมัยใหม่คือ *Klaeber's Beowulf* รุ่นที่สี่โดยอาร์.ดี. ฟุลค์ โรเบิร์ต อี. บยอร์ก และจอห์น ดี. ไนลส์ พิมพ์โดยมหาวิทยาลัยโทรอนโตเมื่อ ค.ศ. 2008 ซึ่งรักษาโครงสร้างของคลาเบอร์ไว้ แต่ปรับปรุงให้สะท้อนงานวิจัยใหม่ว่าด้วยการก่อรูปของตัวเขียนเอกาตถีและปมข้อความที่ยังถกเถียง [9] โดยแนวปฏิบัติการแก้ไขข้อความของฉบับนี้ถูกจัดทำเอกสารอย่างละเอียดในงานของฟุลค์ว่าด้วยการอ่านที่เป็นข้อโต้แย้งในตัวเขียน [10] คู่ขนานระหว่างเคียร์แนนกับฟุลค์จึงแสดงสองขั้วของระเบียบวิธีเอกาตถีครบถ้วน ขั้วหนึ่งมุ่งฟื้นตัวอักษรจากวัตถุด้วยเทคโนโลยี อีกขั้วมุ่งตีความตัวอักษรที่ยังอ่านไม่ได้ด้วยการสันนิษฐานที่มีระเบียบวินัย

### มหากาพย์โรแลนด์: สำนวนเดียวที่แท้จริง

บทกวีมหากาพย์ฝรั่งเศสเรื่อง *La Chanson de Roland* มีสถานะเอกาตถีที่ซับซ้อนกว่า เพราะแม้จะมีตัวเขียนยุคกลางของเรื่องโรแลนด์หลายฉบับ แต่ฉบับอื่นทั้งหมดล้วนเป็นการรจนาขึ้นใหม่หรือสำนวนขยาย มิใช่พยานอิสระของบทกวีเดียวกัน พยานหลักคือสำนวนอ๊อกซฟอร์ดที่ใช้สัญลักษณ์ O อยู่ในทะเบียน Oxford, Bodleian Library, MS Digby 23 ภาคที่สอง จำนวน 72 แผ่นหนังแกะ เขียนด้วยภาษาฝรั่งเศสอ็องโกล-นอร์มันในอังกฤษช่วงปลายศตวรรษที่ 12 ถึงต้นศตวรรษที่ 13 ตามการกำหนดอายุทางอักษรวิทยาของเอ็ม.บี. พาร์กส ตัวเขียนเป็นเล่มรวมที่มีภาคแรกเป็นคำแปลภาษาละตินของคัลคิดิอุสว่าด้วยบทสนทนา *Timaeus* ของเพลโต ประวัติความเป็นเจ้าของยุคกลางไม่ชัดเจน มีจารึกความเป็นเจ้าของของอารามโอเซอนีย์ เข้าสู่หอสมุดของโทมัส อัลเลนผู้ถึงแก่กรรม ค.ศ. 1632 และได้รับบริจาคเข้าห้องสมุดบอดเลเยียนโดยเคนเฮล์ม ดิกบีใน ค.ศ. 1634 [11] ทะเบียนทางการของห้องสมุดบอดเลเยียนยืนยันข้อมูลพื้นฐานเหล่านี้ แม้การดึงหน้าเว็บโดยตรงจะถูกระบบป้องกันบอทขัดขวาง เนื้อหาจึงได้รับการตรวจยืนยันผ่านดัชนีค้นหาทางการเท่านั้น อันเป็นข้อจำกัดที่ควรระบุไว้ตรงไปตรงมา [12]

ข้อเท็จจริงว่าด้วยสำนวนอื่นถูกสรุปไว้ตั้งแต่ศตวรรษที่ 19 โดยทัศนะของเลออง กาติเย่ว่าสำนวนอ๊อกซฟอร์ดสมบูรณ์และเก่าแก่ที่สุด ส่วนสำนวนอื่นทั้งหมดเป็นเพียงสำนวนเขียนใหม่ ไม่ว่าจะเป็นตัวเขียนเวนิสสองเล่ม ตัวเขียนปารีส ฉบับชาโตรู-แวร์ซาย ฉบับลียง เศษหนังสือรงเซวาล หรือบทร้อยกรองนอร์สเรื่อง *Karlomagnus saga* ทั้งหมดนี้เป็นการรจนาใหม่บนฐานของเรื่องเล่า มิใช่สำเนาของบทกวีเดิม [13] ฉบับพิมพ์ครั้งแรกของสำนวนอ๊อกซฟอร์ดโดยฟร็องซิสก์ มีแชล พิมพ์เมื่อ ค.ศ. 1837 หลังการระบุตัวตนของตัวเขียน แต่การชำระยุคศตวรรษที่ 19 ตามธรรมเนียมมักผสานการอ่านจากสำนวนอื่นเข้ากับข้อความหลักอย่างไม่ระบุแหล่งที่มา จนกระทั่งเบดีเยเขียนบทความว่าด้วยอำนาจหน้าที่ของตัวเขียนอ๊อกซฟอร์ดในการจัดตั้งข้อความ ตีพิมพ์ในวารสาร *Romania* เล่มที่ 41 เมื่อ ค.ศ. 1912 ซึ่งพิสูจน์ว่าการอ่านของตัวเขียน O มีคุณค่าเพียงพอที่จะยืนด้วยตนเองโดยไม่ต้องอาศัยการผสมจากสำนวนอื่น [14] แล้วใน ค.ศ. 1922 เบดีเยจึงชำระฉบับของตนในรูปแบบทำสำเนาเทียบต้นอย่างเป็นโปรแกรม พิมพ์ข้อความตามตัวเขียนโดยไม่ปรับปรุงตามรสนิยม อันเป็นคำตอบเชิงวิธีวิทยาสำหรับการชำระข้อความเอกาตถีว่าควรพิมพ์สิ่งที่พยานเพียงฉบับเดียวบอก มิใช่สิ่งที่ผู้ชำระปรารถนาให้มันบอก งานต่อยอดในศตวรรษที่ 20 ยังแสดงว่าพยานเดียวอ่านได้ลึกขึ้นเรื่อย ๆ เช่นการอ่านด้วยแสงอัลตราไวโอเลตของชาร์ล ซามารานเมื่อ ค.ศ. 1929 ซึ่งแก้การอ่านจางหลายจุด และการพิมพ์ภาพถ่ายโฟโตไทป์เต็มเล่มเมื่อ ค.ศ. 1933 [15]

### บทกวีชุดเพิร์ล: พยานเดียวที่เปียกน้ำ

ตัวเขียน London, British Library, Cotton Nero A.x ภาคที่สอง เป็นสมุดหนังแกะขนาดสี่ส่วนจากปลายศตวรรษที่ 14 ที่บรรจุบทกวีสี่เรื่องซึ่งไม่ปรากฏที่อื่นใดเลย ได้แก่ *Pearl*, *Cleanness*, *Patience* และ *Sir Gawain and the Green Knight* เขียนด้วยลายมือเดียวพร้อมภาพเขียนเต็มหน้า งานวิเคราะห์ล่าสุดเสนอว่าภาพร่างใต้สีอาจเป็นฝีมือคนคัดลอกเอง และการวิเคราะห์สีกับหมึกของห้องสมุดบริติชด้วยการถ่ายภาพหลายช่วงคลื่นร่วมกับการเรืองแสงเอกซเรย์ พบว่าเส้นร่างใต้ภาพใช้หมึกตะกั่วที่สอดคล้องกับหมึกของคนคัดลอก อันสนับสนุนข้อเสนอดังกล่าว [16] ตัวเขียนเสียหายจากน้ำและคราบที่ทำให้ตัวอักษรจางลงอย่างเห็นได้ชัด ซึ่งผู้ชำระยุคแรกระบุไว้ในคำนำแล้ว [17] แนวปฏิบัติการชำระของข้อความชุดนี้เป็นตัวอย่างคลาสสิกของการระงับการแก้ไขโดยสันนิษฐาน เพราะเมื่อทุกการอ่านพึ่งพาตัวเขียนเสียหายเล่มเดียว ผู้ชำระจึงเก็บการแก้ไขไว้น้อยที่สุดและระบุทุกครั้งที่แก้ โดยภาคนิทานใช้พื้นที่หลักไปกับตัวอักษรที่จางหรืออ่านไม่ออก มิใช่การเลือกการอ่านระหว่างพยานที่ไม่มีอยู่ ฉบับภาพจำลองของกอลแลนซ์ ค.ศ. 1923 ฉบับชำระอนุรักษ์นิยมของโทลคีนกับกอร์ดอน ค.ศ. 1925 และฉบับมาตรฐานของแอนดรูว์กับวอลดรอน ล้วนดำเนินตามหลักนี้ โครงการดิจิทัล Cotton Nero A.x Project ที่มหาวิทยาลัยแคลการีนำโดยเมอร์เรย์ แม็กกิลลิเวรย์ จัดทำสำเนาดิจิทัลความละเอียดสูงเต็มเล่มเปิดให้สาธารณะ ซึ่งทำให้ผู้ศึกษาตรวจสอบสถานะตัวอักษรได้โดยตรงจากพยานต้นทาง [18]

### มอร์ที ดาร์เธอร์: เมื่อเอกาตถีพิสูจน์ว่าชั่วคราว

กรณีตัวเขียนวินเชสเตอร์ของ *Le Morte Darthur* ของโทมัส มาลอรี ให้บทเรียนตรงข้ามที่สำคัญที่สุด นั่นคือข้ออ้างความเป็นเอกาตถีเป็นข้ออ้างเชิงเงื่อนไขที่พิสูจน์ได้เสมอว่าเป็นเท็จในอนาคต ตั้งแต่ ค.ศ. 1485 ถึง ค.ศ. 1934 วรรณกรรมชุดนี้รู้จักผ่านฉบับพิมพ์ของวิลเลียม แคกสตันเพียงทางเดียว และแม้แต่ฉบับพิมพ์นั้นเองก็หลงเหลือเพียงสองเล่มในโลก ในเดือนมิถุนายน ค.ศ. 1934 วอลเตอร์ เอฟ. โอกส์ช็อตต์ บรรณารักษ์และต่อมาคืออาจารย์ใหญ่โรงเรียนวินเชสเตอร์คอลเลจ ค้นพบระหว่างการจัดทำสารบบว่ามีตัวเขียนกระดาษขนาดใหญ่ราว 480 แผ่นขาดหัวท้าย ซึ่งกลายเป็นที่รู้จักว่า **ตัวเขียนวินเชสเตอร์** เขียนโดยคนเขียนอาชีพสองคนราว ค.ศ. 1470–1483 ปัจจุบันทะเบียน British Library Add. MS 59678 [19]

การเปรียบเทียบสองพยานให้ผลลัพธ์ที่ซับซ้อน รอยหมึกพิมพ์ติดบนตัวเขียนพิสูจน์ว่าเล่มนี้เคยอยู่ในโรงพิมพ์ของแคกสตัน แต่หลักฐานทางภาษาชี้ว่าแคกสตันใช้สำเนาอื่นที่สูญหายไปแล้วเป็นต้นฉบับการพิมพ์ แคกสตันแบ่งเนื้อหาเป็น 21 เล่ม 507 บท ขณะที่ตัวเขียนจัดเป็น 8 นิทาน ทั้งย่อเนื้อหาสงครามโรมันและปรับการสะกดคำไปทางภาษาเคนต์ ตัวอย่างที่โด่งดังคือข้อความพิมพ์ที่ว่า sir Tristram slewe kynge Mark ซึ่งตัวเขียนวินเชสเตอร์ชี้ชัดว่าความเข้าใจนี้ผิด เพราะผู้ล้างแค้นให้บิดาคือเซอร์เบลลิงเจอร์ มิใช่ทริสแตมเป็นผู้สังหารกษัตริย์มาร์ก [20] การค้นพบนี้บีบให้ประวัติศาสตร์การชำระทั้งสายถูกเจรจาใหม่ทันที ยูจีน วีนาเวอร์ชำระข้อความจากตัวเขียนวินเชสเตอร์และเสนอว่างานชิ้นนี้เป็นนิทานแปดเรื่องแยกกัน ไม่ใช่นวนิยายต่อเนื่องตามโครงแบ่งของแคกสตัน ซึ่งจุดประเด็นโต้แย้งกันหลายทศวรรษระหว่างฝ่ายสนับสนุนกับฝ่ายที่ชี้หลักฐานการอ้างอิงข้ามเรื่องภายในงาน [21] บทเรียนวิธีวิทยาที่ชัดเจนคือเมื่อพยานที่สองปรากฏขึ้น ระบบการชำระทั้งระบบต้องถูกรื้อทบทวน ข้ออ้างความเป็นเอกาตถีจึงต้องเขียนไว้เสมอในรูปเงื่อนไขว่ายังไม่มีพยานอื่นปรากฏ มิใช่ไม่มีพยานอื่นอยู่ และโครงการ Malory Project ซึ่งจัดทำฉบับอิเล็กทรอนิกส์กับภาพจำลองดิจิทัลของทั้งตัวเขียนวินเชสเตอร์และฉบับพิมพ์แคกสตันที่เหลือเพียงสองเล่ม แสดงว่าปัญหาพยานเดียวมีอยู่คู่ขนานกันทั้งในวัฒนธรรมตัวเขียนและวัฒนธรรมสิ่งพิมพ์ยุคแรก [22]

### กาตุลลุส: เอกาตถีโดยตัวแทน

บทกวีของกาตุลลุสเป็นกรณีที่แยกชั้นของความเป็นเอกาตถีได้ชัดที่สุด เพราะไม่มีตัวเขียนยุคโบราณหลงเหลือเลย และคลังบทกวีทั้ง 116 บทที่สืบมาถึงเราล้วนสืบสายจากต้นแบบเดียวที่สูญหายไปแล้ว คือ **Codex Veronensis** ซึ่งปรากฏตัวขึ้นที่เมืองเวโรนาราว ค.ศ. 1300 และสูญหายอีกครั้งหลังศตวรรษที่ 14 หลักฐานโดยตรงว่าต้นแบบนี้มีอยู่จริงคือบทกวีเฉลิมฉลองการคืนชีพของกวีกาตุลลุสแห่งเวโรนาโดยเบนเวนูโต กัมเปซานี การระบุว่าต้นแบบที่หายไปคือตัวเขียนที่กัมเปซานีกล่าวถึงเคยถูกตั้งข้อสงสัย แต่ดาเนียล คิสส์ป้องกันไว้เป็นทัศนะข้างมาก [23] ฟรานเซสโก เปตราร์กาเข้าถึงตัวเขียนเวโรนาได้ภายใน ค.ศ. 1347 ก่อนที่มันจะหายไปจากบันทึกทางประวัติศาสตร์ [24]

โครงสร้างพยานที่เหลือจัดระเบียบได้ดังนี้ สัญลักษณ์ T คือทะเบียน Paris, Bibliothèque nationale de France, lat. 8071 จากไต้ศตวรรษที่ 9 เป็นหนังสือรวมบทกวียุคคาโรแล็งเจียนที่เก็บบทที่ 62 เพียงบทเดียว แต่เป็นพยานอิสระนอกสายต้นแบบเวโรนาเพียงฉบับเดียวที่มีอยู่ สัญลักษณ์ O คือทะเบียน Oxford, Bodleian Library, Canonicianus Class. Lat. 30 ศตวรรษที่ 14 น่าจะเขียนที่เวโรนา คัดลอกตรงจากต้นแบบ A ที่สูญแล้ว สัญลักษณ์ G คือทะเบียน Paris, BnF, lat. 14137 เขียนที่เวโรนา ค.ศ. 1375 โดยอันโตนิโอ เดล กาอิโอ ดา เลนนาโก ซึ่งงานของจูเซปเป บิลลาโนวิชระบุตัวผู้คัดลอกได้ มีบทกวีของกัมเปซานีแนบท้ายเล่ม สัญลักษณ์ R คือทะเบียน Vatican, Ottobonianus lat. 1829 เขียนที่ฟลอเรนซ์ราว ค.ศ. 1375–1392 ให้แก่โคลุชโช ซาลูตาติ ซึ่งเพิ่มการแก้ไขโดยสันนิษฐานของตนเองลงในเล่ม G และ R เป็นตัวเขียนพี่น้องคัดจากต้นฉบับกลาง X ที่สูญหายซึ่งคัดจาก A อีกทอด สัญลักษณ์ m คือทะเบียน Venice, Marcianus lat. 12.80 ราว ค.ศ. 1400 เป็นสำเนาของ R เพียงเท่านั้นจึงไม่เพิ่มข้อมูลใหม่ใด ๆ นอกจากนี้ยังมีตัวเขียน S หรือ Codex Sannazarii ที่สูญหายไปแล้วซึ่งปาร์ราซีโอเคยเห็นราวต้นศตวรรษที่ 16 ใกล้เมืองลียง โดยการอ่านของมันหลงเหลือในขอบอรรถาธิบายของปาร์ราซีโอเอง เป็นเครื่องเตือนว่าญาติที่สูญไปของพยานเดียวอาจโผล่กลับมาผ่านขอบหนังสือได้เสมอ มีตัวเขียนยุคมนุษยนิยมอีกราว 129 เล่มทั่วยุโรปซึ่งทั้งหมดเป็นสำเนาลูกหลานที่ไม่เพิ่มพูนอะไร [25]

สถานการณ์เช่นนี้เรียกว่าเอกาตถีโดยตัวแทน เพราะทุกพยานที่มีอยู่ล้วนสืบจากต้นแบบเดียวที่เสียไปแล้ว การสอบเทียบระหว่าง O กับ G กับ R ยังทำได้และมีประโยชน์ แต่มันสอบเทียบความผิดพลาดของสำเนารุ่นหลังเท่านั้น มิได้เทียบพ้นต้นแบบที่เสียไป ดังนั้นเมื่อต้นแบบ A มีข้อความเสียหายหรือคลาดเคลื่อน ความเสียหายนั้นถูกถ่ายทอดลงทุกสำเนาโดยไม่มีพยานใดข้ามมันไปได้ ยกเว้น T ที่เหลือบทเดียว สายฉบับพิมพ์จึงต้องพึ่งการแก้ไขโดยสันนิษฐานอย่างหลีกเลี่ยงไม่ได้ ตั้งแต่ลัมบินุส ค.ศ. 1566 ที่เริ่มแก้แบบเป็นระบบ สคาลิเจอร์ ค.ศ. 1577 จนถึงไมเนอร์สในชุด Oxford Classical Texts ค.ศ. 1958 ที่ยึด O เป็นหลัก และคิสส์เองในฉบับชำระ ค.ศ. 2008 พร้อมโครงการดิจิทัล Catullus Online ที่จัดทำข้อมูลตัวเขียนและข้อความชำระเปิดให้สาธารณะ [26] บริบทยุคมนุษยนิยมเวโรนาที่กวินโน ดา เวโรนาเป็นแกนกลางก็ได้รับการจัดทำเอกสารอย่างเป็นระบบเช่นกัน อันเป็นเครือข่ายที่สำเนาสายเวโรนาหมุนเวียนอยู่ในต้นศตวรรษที่ 15 [27] กรณีกาตุลลุสจึงสอนว่าความเป็นเอกาตถีไม่ใช่คุณสมบัติแบบเปิดปิดของตัวเขียนเล่มเดียว แต่เป็นคุณสมบัติของระบบสายสืบทั้งระบบว่ามีจุดคอขวดอยู่ตรงไหน

### Codex Regius: พยานเดียวที่มีช่องว่างใบขาด

คลังบทกวีนอร์สยุคกลางที่เรียกว่า *Poetic Edda* พึ่งพาตัวเขียนเดียวเกือบทั้งหมด คือทะเบียน Reykjavík, Árni Magnússon Institute, GKS 2365 4to หรือ **Codex Regius** หรือ **Konungsbók** เป็นหนังแกะ 45 แผ่น เขียนด้วยลายมือเดียวของคนไอซ์แลนด์ กำหนดอายุทางอักษรวิทยาราว ค.ศ. 1270–1280 เป็นแหล่งเดียวหรือแหล่งหลักของบทกวีเอดดา 29 บท แบ่งเป็น 10 บทว่าด้วยเทพเจ้าและ 19 บทว่าด้วยวีรบุรุษ บท *Hávamál* มีอยู่ในตัวเขียนนี้เพียงฉบับเดียวในโลก ส่วน *Völuspá* มีฉบับขนานเพิ่มใน *Hauksbók* ประวัติเล่มเริ่มเมื่อบิชอปบรินโยลฟูร์ สไวนส์สสันได้รับเมื่อ ค.ศ. 1643 และถวายแด่พระเจ้าเฟรเดริกที่สามแห่งเดนมาร์กเมื่อ ค.ศ. 1662 จึงได้ชื่อว่าเป็นหนังสือของกษัตริย์ อยู่ห้องสมุดหลวงโคเปนเฮเกนจนถึงวันที่ 21 เมษายน ค.ศ. 1971 ที่ถูกส่งกลับไอซ์แลนด์ทางเรือ และผูกเล่มใหม่ด้วยหนังลูกวัวเมื่อ ค.ศ. 1992 ทะเบียนทางการบนระบบ handrit.is เป็นแหล่งอ้างอิงมาตรฐานของเล่มนี้ แม้เซิร์ฟเวอร์จะตอบปฏิเสธโปรแกรมดึงอัตโนมัติ สถานะแหล่งอ้างอิงทางการยังคงอยู่ และมีภาพถ่ายโฟโตไทป์เต็มเล่มพิมพ์ไว้ตั้งแต่ ค.ศ. 1891 [28] นักวิชาการสรุปสถานะของเล่มนี้ไว้ตรง ๆ ว่าเมื่อนักวิชาการพูดถึงบทกวีเอดดา สิ่งที่ถูกหมายถึงโดยทั่วไปคือเนื้อหาของตัวเขียนเพียงเล่มเดียว [29]

จุดสำคัญของกรณีนี้คือปัญหา **ช่องว่างใบขาด** แผ่นรวมที่ห้าราว 8 แผ่นหรือ 16 หน้าหายไประหว่างท้ายบท *Sigrdrífumál* กับต้นบท *Brot af Sigurðarkviðu* บรินโยลฟูร์เองกำกับรอยต่อนี้ไว้ด้วยภาษากรีกว่าเป็นเรื่องไร้หัว เอ็ยนาร์ เจ. เปตูร์สสันเสนอข้อสังเกตที่ละเอียดอ่อนว่า เนื่องจากบท *Sigrdrífumál* ฉบับสมบูรณ์มีอยู่ในสำเนากระดาษศตวรรษที่ 17 ซึ่งเชื่อมกับการคัดลอก *Völsunga saga* ที่บรินโยลฟูร์ได้รับหนังแกะต้นฉบับเมื่อ ค.ศ. 1641 ช่องว่างจึงเกิดขึ้นทางกายภาพระหว่าง ค.ศ. 1641 ถึง 1643 คือแผ่นหลุดหายก่อนการได้รับเพียงไม่กี่ปี [30] การสืบสร้างเนื้อหาที่ขาดหายพึ่งแหล่งทางอ้อมสามชั้น ได้แก่ สำเนากระดาษศตวรรษที่ 17 สำหรับท้ายบท *Sigrdrífumál* การเล่าความร้อยแก้วใน *Völsunga saga* ซึ่งใช้หนังแกะเล่มนี้หรือต้นฉบับของมันขณะที่แผ่นยังอยู่ครบ และบทที่ถูกอ้างหรือเล่าใหม่ใน *Snorra Edda* ของสนอร์รี สตูร์ลูสสัน พร้อมฉบับขนานของ *Völuspá* ใน *Hauksbók* ผู้ชำระตั้งแต่บุกเกอเป็นต้นมา จนถึงเนคเคลกับคูน และฉบับสองภาษาเผยแพร่เปิดของเพตติต ล้วนกำกับเนื้อหาที่เติมจากประเพณีทางอ้อมไว้อย่างชัดเจน ห้ามเติมเงียบ ๆ ราวกับว่ามันมาจากพยานหลัก [31] หลักปฏิบัตินี้คือการติดฉลากประเพณีทางอ้อม ซึ่งเป็นข้อบังคับจริยธรรมของการชำระข้อความเอกาตถีที่มีช่องว่าง

## 4.3 เอกสารที่ถูกทำลายและสำเนาแทน

### อัสแซร์: พยานที่ไหม้แล้ว แต่ข้อความรอด

กรณีที่รุนแรงกว่าการมีพยานเสียหายเล่มเดียวคือการไม่มีพยานกายภาพเหลืออยู่เลย ชีวประวัติ *Vita Ælfredi regis* ของอัสแซร์รู้จักจากตัวเขียนเพียงฉบับเดียว คือ Cotton MS Otho A.xii เขียนราว ค.ศ. 1000 ซึ่งถูกไฟเผาในเพลิงไหม้บ้านแอชเบิร์นฮัมวันที่ 23 ตุลาคม ค.ศ. 1731 เพลิงเดียวกันที่ลวกเบโอวุลฟ์ ปัจจุบันไม่มีส่วนใดของตัวเขียนหลงเหลือ ข้อความถูกสืบสร้างจากพยานแทนสองชั้น ชั้นแรกคือสำเนาถอดความก่อนไฟไหม้หลายชุดหลายมือ ซึ่งบางสำเนาคัดลอกคำอรรถาธิบายริมขอบของแมทธิว พาร์กเกอร์ปะปนไปกับข้อความหลักราวกับเป็นส่วนหนึ่งของต้นฉบับ เป็นอันตรายเชิงการชำระที่ผู้ศึกษาต้องระวังเป็นพิเศษ และมีภาพจำลองหน้าแรกของตัวเขียนทำไว้เมื่อ ค.ศ. 1722 ก่อนไฟไหม้เพียงเก้าปี ชั้นที่สองคือเนื้อหาที่ผู้เขียนยุคกลางคัดมาฝังไว้ในงานของตน ได้แก่ เบิร์ชเฟิร์ทแห่งแรมซีย์ใน *Historia Regum* ฟลอเรนซ์แห่งวุร์เซสเตอร์ และผู้รจนา *Encomium Emmae Reginae* น่าสนใจว่าเล่ม Otho A.xii เดียวกันยังเคยบรรจุบท *The Battle of Maldon* ซึ่งปัจจุบันมีอยู่ผ่านสำเนาของแคสลีย์เท่านั้น แสดงว่าการสูญพยานครั้งเดียวอาจกวาดข้อความหลายเรื่องไปพร้อมกัน [32] [33]

งานพิมพ์ของจอห์น สมิธ ค.ศ. 1721 อาศัยตัวเขียนที่ยังอยู่ก่อนไฟไหม้เพียงสิบปี ขณะที่ฉบับมาตรฐานของวิลเลียม เฮนรี สตีเฟนสัน ค.ศ. 1904 สร้างขึ้นโดยการสอบเทียบสำเนาที่รอดทั้งหมดเข้าด้วยกัน พร้อมคำนำยาวราว 130 หน้า และยังเป็นข้อความละตินมาตรฐานจนถึงทุกวันนี้ [34] การถกเถียงความจริงแท้ของงานเขียนนี้ดำเนินไปเป็นร้อยปี ตั้งแต่การโจมตีในศตวรรษที่ 19 ของไรต์กับโฮเวิร์ท ผ่านข้อเสนอของกัลเบรธใน ค.ศ. 1964 ว่าเป็นของปลอมยุคหลัง และข้อเสนอของสไมธ์ว่าเบิร์ชเฟิร์ทเป็นผู้แต่ง ไปจนถึงการป้องกันของพลัมเมอร์ที่ชี้ว่างานต้องมีอายุไม่เกิน ค.ศ. 974 ของไวต์ล็อกที่สรุปว่าเป็นอัสแซร์แท้ และของสตีเฟนสันเอง จนปัจจุบันข้อความได้รับการยอมรับว่าแท้เกือบเป็นเอกฉันท์ [35] ประเด็นวิธีวิทยาที่สำคัญคือเมื่อพยานกายภาพถูกทำลาย การถกเถียงความจริงแท้ทั้งหมดต้องเลื่อนไปตั้งอยู่บนคุณภาพของสำเนาแทน การวิจารณ์ภายในด้านสไตล์ คำศัพท์ และความไม่สอดคล้องยุคสมัย และการอ้างข้ามกับงานยุคกลางอิสระที่คัดเนื้อหามาไว้ ซึ่งเป็นวิธีที่การถกเถียงถูกดำเนินไปจริงตลอดหนึ่งร้อยปี กรณีนี้จึงเป็นต้นแบบว่า **สำเนาแทนของเอกสารที่ถูกทำลายกลายเป็นหลักฐานปฐมภูมิ** ในทางปฏิบัติ เพราะไม่มีอะไรอื่นให้เทียบอีกแล้ว

### แป้นเขียนอาร์คิมิดีส: จากภาพถ่ายปี 1906 สู่การอ่านใหม่ทั้งหมด

กรณีแป้นเขียนอาร์คิมิดีสแสดงว่าพยานเอกาตถีสามารถให้ข้อความใหม่ได้เกือบหนึ่งศตวรรษหลังฉบับชำระมาตรฐาน เพียงด้วยเทคโนโลยีภาพและนโยบายเปิดข้อมูล ตัวเขียนนี้เป็นหนังสือสวดมนต์ไบแซนไทน์ศตวรรษที่ 13 ที่ผลิตจากหนังแกะรีไซเคิลจากสำเนาคอนสแตนติโนเปิลศตวรรษที่ 10 ซึ่งเกี่ยวข้องกับสำนักของลีโอนนักเรขาคณิต บรรจุบทความเจ็ดเรื่องของอาร์คิมิดีส รวมทั้ง *The Method of Mechanical Theorems* และ *Stomachion* ที่ไม่มีสำเนาอื่นในโลกเลย และเป็นแหล่งภาษากรีกเพียงแหล่งเดียวของบท *On Floating Bodies* นอกจากนี้ยังมีคำปราศรัยของไฮเปอริดีสที่เกือบสูญ และคำอรรถาธิบายว่าด้วย *Categories* ของอริสโตเติลถูกเขียนทับอยู่ด้วย หนังสือสวดมนต์เสร็จสมบูรณ์ 13 เมษายน ค.ศ. 1229 ตามคำจารึกปีที่จอห์น โลว์เดนอ่านออกด้วยแสงอัลตราไวโอเลตเมื่อ ค.ศ. 2002 น่าจะผลิตที่เยรูซาเลมตามลักษณะบทสวด เคยเป็นของอารามมาร์ซาบาตามจารึกศตวรรษที่ 16 ที่ปัจจุบันสูญหาย ก่อนอยู่ที่เมโทเชียนของพระศาสนสถานแห่งไม้กางเขนในคอนสแตนติโนเปิลราว ค.ศ. 1840 [36]

จุดเปลี่ยนสองครั้งกำหนดประวัติศาสตร์ยุคใหม่ของเล่มนี้ ครั้งแรกเมื่อพาปาโดปุโลส-เคราเมอุสจัดทำสารบบเมื่อ ค.ศ. 1899 และถอดความข้อความใต้ไม่กี่บรรทัด ซึ่งนำไปสู่การเดินทางของโยฮัน ลูทวิช ไฮเบิร์กไปดูตัวเขียนใน ค.ศ. 1906 ไฮเบิร์กถ่ายภาพตัวเขียนและรวมผลไว้ในฉบับชำระใหม่ทั้งชุดที่พิมพ์ ค.ศ. 1910–1915 แผ่นหนึ่งที่หลุดออกผ่านมือทิชเอนดอร์ฟปัจจุบันอยู่ที่ห้องสมุดมหาวิทยาลัยเคมบริดจ์โดยไนเจล วิลสันระบุตัวตนได้เมื่อ ค.ศ. 1968 ครั้งที่สองเมื่อตัวเขียนถูกขายที่ประมูลคริสตีส์ในนิวยอร์กวันที่ 29 ตุลาคม ค.ศ. 1998 ในราคาสองล้านดอลลาร์หลังคำสั่งศาลของฝ่ายกรีกไม่สำเร็จ แล้วถูกฝากไว้ที่พิพิธภัณฑ์ศิลปะวอลเทอรส์ในเมืองบอลติมอร์เมื่อมกราคม ค.ศ. 1999 สิ่งที่ตามมาคือการอนุรักษ์ การถ่ายภาพหลายช่วงคลื่นโดยทีมของโรเจอร์ อีสตัน คีธ นอกซ์ และวิลเลียม คริสเตนส์-แบร์รี ตามด้วยการสแกนเรืองแสงเอกซเรย์ ซึ่งฟื้นข้อความใต้ที่ไฮเบิร์กอ่านไม่เห็น และข้อมูลภาพดิบทั้งหมดถูกเปิดเผยต่อสาธารณะเป็นระบบ [37] ความเสียหายที่เพิ่มขึ้นหลัง ค.ศ. 1908 ยังสอนข้อเท็จจริงอีกด้าน ได้แก่ หน้าที่มีข้อความอาร์คิมิดีสสามหน้าหายไปทั้งหมดแม้ไฮเบิร์กเคยถอดความไว้ ความเสียหายจากเชื้อราอย่างหนัก และภาพนักเทศน์สี่ภาพที่วาดทับข้อความหลัง ค.ศ. 1929 [38] ฉบับชำระวิชาการโดยเนทซ์ โนเอล เชอร์เนตสกา และวิลสัน จัดพิมพ์เป็นสองเล่มโดยมหาวิทยาลัยเคมบริดจ์เพื่อพิพิธภัณฑ์วอลเทอรส์ [39] พร้อมหนังสือเล่าเรื่องเชิงประชานิยมของเนทซ์กับโนเอลที่เผยแพร่กรณีนี้สู่ผู้อ่านวงกว้าง [40] ความหมายเชิงระเบียบวิธีคือเมื่อข้อความอยู่ใต้ข้อความอื่นบนพยานเดียว การอ่านซ้ำทางวิทยาศาสตร์ไม่ใช่เครื่องมือเสริม แต่เป็นประตูเดียวที่เปิดไปยังข้อความที่มิฉะนั้นจะไม่มีใครเห็นอีกเลย

## 4.4 เครื่องมือวิทยาศาสตร์และการอ่านซ้ำ

ชุดเทคนิคที่ใช้อ่านพยานเอกาตถีซ้ำจัดกลุ่มได้เป็นสี่ประเภทหลัก ประเภทแรกคือ **การถ่ายภาพหลายช่วงคลื่น** ซึ่งถ่ายภาพหน้าตัวเขียนที่หลายความยาวคลื่นแล้วประมวลผลเป็นชั้นข้อมูลรวม เพื่อแยกลายเซ็นสเปกตรัมของหมึกแต่ละชนิดออกจากหนังแกะและหมึกที่เขียนทับ เป็นหัวใจของการฟื้นข้อความใต้แป้นเขียนอาร์คิมิดีส [41] ประเภทที่สองคือ **แสงอัลตราไวโอเลต** สำหรับอักษรที่ถูกลบหรือจาง มีตัวอย่างตั้งแต่การอ่านตัวเขียนโรแลนด์ของซามารานเมื่อ ค.ศ. 1929 การอ่านคำจารึกปี 1229 ของโลว์เดน จนถึงการอ่านกว่า 100 จุดในตัวเขียนเบโอวุลฟ์ [42] ประเภทที่สามคือ **การเรืองแสงเอกซเรย์** ซึ่งทำแผนที่ธาตุเคมีบนพื้นผิวเพื่อแยกข้อความใต้และวิเคราะห์องค์ประกอบสี ใช้ทั้งกับแป้นเขียนอาร์คิมิดีสและการวิเคราะห์สีภาพเขียนในตัวเขียนเพิร์ล [43] ประเภทที่สี่คือ **การลงวันที่ด้วยคาร์บอนกัมมันตรังสี** ของวัสดุรองรับการเขียน ซึ่งล่าสุดถูกบูรณาการกับการวิเคราะห์รูปร่างอักษรด้วยปัญญาประดิษฐ์อย่างเป็นระบบ

งานของโปโปวิชกับคณะตีพิมพ์ในวารสาร *PLoS One* เมื่อ ค.ศ. 2025 เป็นแบบแผนล่าสุดของการทำงานแบบสามเหลี่ยมหลักฐาน ทีมวิจัยใช้การหาอายุคาร์บอนกัมมันตรังสีด้วยวิธีสเปกโตรเมทรีมวลอะตอมบนหนังแกะ พร้อมการล้างสารไขมันก่อนวิเคราะห์เพื่อกันข้อผิดพลาดจากสารปนเปื้อน และระวังปัญหาการหาอายุที่ให้ช่วงความน่าจะเป็นสองยอดจากเส้นโค้งการหาอายุ ซึ่งจัดการด้วยวิธีเบย์เซียน ฝ่ายการวิเคราะห์รูปร่างอักษรใช้แบบจำลองปัญญาประดิษฐ์ที่ฝึกจากตัวอย่างที่หาอายุด้วยคาร์บอนกัมมันตรังสีแล้วเพียง 24 ตัวอย่าง โดยจัดคุณลักษณะรูปร่างตัวอักษรแยกจากเนื้อหาข้อความโดยเจตนา ผลทำนายมีความคลาดเคลื่อนสัมบูรณ์เฉลี่ยราว 28–31 ปี และเมื่อทำนายตัวเขียนที่ไม่มีวันที่อีก 135 ฉบับ ผลสอดคล้องกับการประเมินหลังของนักอักษรวิทยาผู้เชี่ยวชาญราว 79 เปอร์เซ็นต์ ข้อสรุปเชิงวิธีวิทยาที่สำคัญที่สุดคือการบูรณาการหลักฐานอิสระสามชั้น ได้แก่ ชั้นวัสดุจากคาร์บอนกัมมันตรังสี ชั้นเรขาคณิตจากการวิเคราะห์รูปร่าง และชั้นการตัดสินของนักอักษรวิทยาผู้เชี่ยวชาญ โดยระบุความไม่แน่นอนของแต่ละชั้นไว้อย่างชัดเจนและมองเห็นได้ มิใช่รายงานวันที่เดียวที่ดูลอย ๆ [44] สำหรับพยานเอกาตถีที่ไม่มีตัวเขียนอื่นเทียบเพื่อลงวันที่เชิงสัมพัทธ์ ระเบียบแบบสามเหลี่ยมนี้จึงเป็นทางเลือกที่มีวินัยที่สุดในปัจจุบัน

อีกหลักการหนึ่งที่กรณีศึกษาทั้งหมดยืนยันตรงกันคือ **การอนุรักษ์สร้างหลักฐานและทำลายหลักฐานไปพร้อมกัน** กรอบกระดาษของปี 1845 ช่วยชะลอการเสื่อมสลายของแผ่นเบโอวุลฟ์แต่ปิดบังตัวอักษรขอบแผ่นไปกว่าหนึ่งศตวรรษครึ่ง กว่าจะถูกอ่านคืนด้วยแสงส่องหลังแบบดิจิทัลซึ่งกู้ข้อมูลโดยไม่ต้องแตะต้องวัตถุ การผูกเล่มใหม่ของ Codex Regius ในยุคก่อนหน้าถูกวิจารณ์ว่าทำร้ายขอบหนังแกะ ก่อนการผูกเล่มเพื่อการอนุรักษ์ใน ค.ศ. 1992 จะมาแก้สถานการณ์ ภาพนักเทศน์ที่วาดทับแป้นเขียนอาร์คิมิดีสหลัง ค.ศ. 1929 คือการอนุรักษ์เชิงศิลปะที่ทำลายข้อความโดยตรง บทเรียนร่วมคือทุกการแตะต้องวัตถุต้องถูกถ่วงดุลระหว่างประโยชน์การอ่านในปัจจุบันกับข้อมูลที่อาจสูญถาวร และเทคนิคที่ดีที่สุดคือเทคนิคที่กู้คืนข้อมูลโดยไม่แตะต้องวัตถุเลย

หลักการปิดท้ายคือ **สำเนาดิจิทัลเปิดเป็นหน้าที่ของการเอาใจใส่** โครงการอิเล็กทรอนิกส์เบโอวุลฟ์เปิดภาพหลายช่วงคลื่นพร้อมสำเนายุคแรกทั้งหมด [45] โครงการตัวเขียน Cotton Nero A.x ที่แคลการีเปิดภาพความละเอียดสูงเต็มเล่ม Malory Project เปิดภาพจำลองทั้งตัวเขียนวินเชสเตอร์และฉบับพิมพ์แคกสตัน Catullus Online เปิดข้อมูลตัวเขียนทั้งสาย ระบบ handrit.is เปิดทะเบียนมาตรฐานของคลังตัวเขียนไอซ์แลนด์ และข้อมูลภาพดิบของแป้นเขียนอาร์คิมิดีสถูกปล่อยเป็นสาธารณะทั้งหมด เหตุผลมีสองชั้น ชั้นแรกคือการลดความเสี่ยงจากการหยิบจับวัตถุเอกาตถีที่ไม่อาจทดแทน ชั้นที่สองคือการกระจายโอกาสตรวจสอบพยานไปสู่ผู้ศึกษาทุกที่ เพราะเมื่อพยานมีฉบับเดียว การอ่านที่ผิดพลาดเพียงครั้งเดียวอาจกลายเป็นความเชื่อมั่นร่วมของวงการได้หากไม่มีใครตรวจสอบย้อนได้ หลักการอนุรักษ์และเข้าถึงคู่ขนานกันนี้ยังเป็นแกนของโครงการความทรงจำแห่งโลกของยูเนสโกที่จะปรากฏบทบาทอีกครั้งในหัวข้อถัดไป

## 4.5 ปมความจริงแท้ของเอกาตถีไทย

ปัญหาพยานเดียวไม่ใช่เรื่องของโลกตะวันตกเพียงฝ่ายเดียว กรณีไทยที่เข้มข้นที่สุดคือ **จารึกหลักที่ 1** หรือ **ศิลาจารึกพ่อขุนรามคำแหง** เป็นศิลาจารึกหินทรายสี่ด้าน ทะเบียนพิพิธภัณฑสถานแห่งชาติกรุงเทพ เลขที่ 158/2511 ตามธรรมเนียมลงวันที่ ค.ศ. 1292 ตรงกับศักราชศก 1211 ค้นพบเมื่อ ค.ศ. 1833 ณ เนินปราสาท เมืองสุโขทัย โดยพระบาทสมเด็จพระจอมเกล้าเจ้าอยู่หัวขณะยังดำรงพระอิสริยยศเจ้าชายมงกุฎ พร้อมพระแท่นหินหลักเดียวกัน จารึกหลักนี้ได้รับการนับถือเป็นรากฐานของอักษรไทยและประวัติศาสตร์นิพนธ์ไทยสมัยใหม่ งานศึกษาอ้างอิงมาตรฐานด้านอักษรศาสตร์คือบทวิเคราะห์จารึกฉบับเต็มโดยกริสโวลด์กับประเสริฐ ณ นคร ตีพิมพ์ในวารสารสยามสมาคมเมื่อ ค.ศ. 1971 ซึ่งจัดทำคำถอดอักษร คำแปล และบทวิเคราะห์ด้านอักษรวิทยาอย่างเป็นระบบ [46] สถานะเชิงสถาบันสูงสุดในระดับโลกคือการขึ้นทะเบียนเป็นมรดกความทรงจำแห่งโลกของยูเนสโกเมื่อ ค.ศ. 2003 โดยเอกสารเสนอชื่อระบุบทบาทของจารึกต่อการสืบทอดอักษรไทย การอนุรักษ์ และการเข้าถึง ทั้งจารึกยังถูกใช้เป็นหลักฐานหลักในการเสนอชื่อเมืองประวัติศาสตร์สุโขทัยขึ้นเป็นมรดกโลกสำเร็จเมื่อ ค.ศ. 1991 อันสะท้อนกระแสประวัติศาสตร์นิพนธ์สมัยใหม่ที่วางยุคสุโขทัยไว้เป็นยุคทองของประวัติศาสตร์ไทย [47]

ความเป็นเอกาตถีของจารึกหลักที่ 1 มีลักษณะพิเศษต่างจากกรณีตัวเขียนทั้งหมดข้างต้น ตัวเขียนยุคกลางเป็นสำเนาของข้อความที่เคยมีต้นฉบับก่อนหน้า แต่จารึกคือวัตถุและข้อความในเวลาเดียวกัน เมื่อความจริงแท้ของวัตถุถูกตั้งข้อสงสัย ความจริงแท้ของข้อความทั้งหมดก็สั่นคลอนไปพร้อมกัน การถกเถียงเชิงวิชาการปะทุขึ้นเป็นระลอกในทศวรรษ 1980 เริ่มจากงานของนิธิ เอียดสระบุรีที่ตั้งข้อสงสัยบริบททางประวัติศาสตร์ในช่วงกลางทศวรรษ 2520 ซึ่งกระตุ้นวงสนทนาทางปัญญาในประเทศจนมีการจัดสัมมนาและรวบรวมบทความเป็นเล่มพิเศษเมื่อ ค.ศ. 1988 และต่อมามีงานวิเคราะห์สังคมวิทยาของการถกเถียงนี้โดยตรง [48] ระลอกสากลเริ่มจากบทความของไมเคิล วิกเกอรี นำเสนอที่ประชุมในออสเตรเลียเมื่อ ค.ศ. 1987 และขยายเป็นบทเปิดของเล่มรวมบทความว่าด้วยข้อถกเถียงจารึกพ่อขุนรามคำแหงที่เจมส์ อาร์. แชมเบอร์ลินบรรณาธิการ จัดพิมพ์โดยสยามสมาคมเมื่อ ค.ศ. 1991 ซึ่งรวมผู้เขียนทั้งสองฝ่ายไว้ครบ ทั้งวิกเกอรี ปิริยะ ไกรฤกษ์ เทอร์วีล ดิลเลอร์ วุดเวิร์ด ไวแอตต์ กอสลิง เพนท์ เจดนีย์ ดอเร และโคนอร์ โดยเนื้อหาเต็มของเล่มเปิดให้อ่านออนไลน์ [49]

ข้อโต้แย้งฝ่ายตั้งข้อสงสัยความจริงแท้มีหลายชั้น วิกเกอรีซึ่งตั้งชื่อบทความเชิงเปรียบเทียบว่าเป็นกะโหลกพิลท์ดาวน์แห่งประวัติศาสตร์เอเชียตะวันออกเฉียงใต้ โจมตีด้วยหลักฐานทางอักษรวิธีว่าระบบวรรณยุกต์ในจารึกครบถ้วนตามแบบสมัยใหม่ ซึ่งมีหลักฐานยืนยันเป็นครั้งแรกในเอกสารศตวรรษที่ 17 ถึง 18 เท่านั้น ตามด้วยข้อสังเกตด้านคำศัพท์และเนื้อหา และต่อมาได้ขยายข้อโต้แย้งต่อในวารสารสยามสมาคมเมื่อ ค.ศ. 1995 [50] ฝ่ายเดียวกัน ปิริยะ ไกรฤกษ์ เขียนบทอวสานของเล่มรวมบทความเสนอว่าศิลานี้ถูกจารึกขึ้นในสมัยพระบาทสมเด็จพระจอมเกล้าเจ้าอยู่หัวเอง อันยกระดับข้อถกเถียงจากเรื่องอายุของข้อความไปสู่เรื่องผู้ผลิตวัตถุ [51]

ฝ่ายสนับสนุนความจริงแท้โต้แย้งจากหลายมุม ได้แก่ การวิเคราะห์อักษรศาสตร์และประวัติศาสตร์ฉบับเต็มของกริสโวลด์กับประเสริฐที่กล่าวถึงแล้ว [52] บทความของเดวิด เค. ไวแอตต์ที่รวบรวมข้อโต้แย้งเชิงบริบทประวัติศาสตร์ [53] งานของไฮแรม วุดเวิร์ดที่ชี้ความสอดคล้องของสถาปัตยกรรมศาสนาสุโขทัยกับเนื้อหาจารึก [54] บทความของฮันส์ เพนท์ที่วิเคราะห์ประเด็นยากของจารึกหลักที่ 1 ในวารสารสยามสมาคมเมื่อ ค.ศ. 1989 [55] และงานสัทวิทยาประวัติศาสตร์ของแอนโทนี ดิลเลอร์ที่ทดสอบข้อโต้แย้งด้านภาษา [56] งานประเมินภายหลังของบี.เจ. เทอร์วีลใช้หลักมีดโกนอ็อกแฮมชั่งน้ำหนักสมมติฐานทั้งสองฝ่าย และตีพิมพ์หนังสือประเมินประเด็นนี้โดยเฉพาะใน ค.ศ. 2010 [57] สถานะของการถกเถียง ณ ปัจจุบันคือยังไม่มีการวินิจฉัยที่ทุกฝ่ายยอมรับอย่างเป็นทางการ แต่ทัศนะข้างมากของวงวิชาการยอมรับความจริงแท้โดยสาระ และการขึ้นทะเบียนของยูเนสโกเมื่อ ค.ศ. 2003 สะท้อนการยอมรับเชิงสถาบันระหว่างประเทศ

ด้านหลักฐานทางวิทยาศาสตร์ การทดสอบเชิงวัสดุเมื่อ ค.ศ. 1990 ใช้กล้องจุลทรรศน์อิเล็กตรอนแบบสแกนร่วมกับการวิเคราะห์รังสีเอกซ์แบบกระจายพลังงาน เปรียบเทียบสภาพผุกร่อนของจารึกหลักที่ 1 กับจารึกสมัยสุโขทัยอีกสี่หลัก สรุปว่าอายุการผุกร่อนใกล้เคียงกันในช่วงราว 700 ถึง 500 ปี ข้อสรุปนี้ถูกกล่าวถึงอย่างแพร่หลายในงานรายงานรอง แต่รายงานการทดสอบฉบับเต็มที่ตรวจสอบได้โดยตรงยังไม่ได้รับการบันทึกไว้ในชุดข้อมูลการสืบค้นของงานเขียนนี้ จึงต้องกำกับสถานะไว้ตรงไปตรงมาว่าเป็นหลักฐานที่รับรู้ผ่านการรายงานรองเท่านั้น [58] บทเรียนวิธีวิทยาที่สำคัญที่สุดจากปมนี้จึงไม่ใช่คำตอบว่าจารึกแท้หรือไม่แท้ แต่คือการแยกชั้นคำถามออกจากกันอย่างเด็ดขาด คำถามทางวัตถุว่าหินมีอายุเท่าใดและผุกร่อนสม่ำเสมอกับหลักอื่นหรือไม่ เป็นคำถามที่เครื่องมือวิทยาศาสตร์ตอบได้ ส่วนคำถามทางอักษรศาสตร์ว่าระบบวรรณยุกต์และคำศัพท์สอดคล้องกับสายวิวัฒนาการภาษาหรือไม่ เป็นคำถามเชิงอักษรศาสตร์ที่ต้องตอบด้วยการเทียบเคียงกับคลังจารึกทั้งชุด การหลอมสองคำถามนี้เข้าด้วยกันทำให้การถกเถียงเลื่อนลอยไปโดยไม่มีทางระบาย และกรณีนี้ยังยืนยันข้อสังเกตเชิงเปรียบเทียบว่าจารึกไทยส่วนใหญ่เป็นเอกาตถีรายหลัก คลังจารึกรวมจึงทำหน้าที่แทนการสอบเทียบฉบับ ด้วยการเทียบอักษรและภาษาข้ามหลักผ่านฐานข้อมูลจารึกในประเทศไทยซึ่งเปิดให้สาธารณะ [59]

## 4.6 การแก้อักษรโดยสันนิษฐานกับเอกาตถี

เมื่อการสอบเทียบฉบับใช้ไม่ได้ การแก้ไขข้อความที่เชื่อว่าเสียหายจึงตกอยู่กับ **การแก้อักษรโดยสันนิษฐาน** (*conjectural emendation*) คือการเสนอตัวอักษรหรือคำแทนที่ตัวอักษรหรือคำที่อ่านไม่ได้หรือให้ความหมายไร้สาระ โดยอาศัยหลักภาษา บริบท มิติ หรือสำนวนขนาน แทนที่จะอาศัยพยานอื่น คู่มืออ้างอิงมาตรฐานว่าด้วยการวิจารณ์ข้อความกรีกละตินของสำนักพิมพ์มหาวิทยาลัยอ๊อกซฟอร์ดสรุปสถานะของเทคนิคนี้ไว้สองข้างพร้อมกัน กล่าวคือ เนื่องจากไม่มีต้นแบบใดปราศจากข้อผิดพลาด การแก้อักษรโดยสันนิษฐานจึงจำเป็น แต่ขณะเดียวกันก็ขาดฐานวิธีวิทยาที่ชัดเจนและประกอบด้วยการตัดสินใจเฉพาะกรณีเป็นส่วนใหญ่ แนวปฏิบัติที่รับผิดชอบจึงอยู่ตรงกลางระหว่างการยอมรับทุกการอ่านที่อ่านได้ความ กับการแก้ไขทุกครั้งที่จินตนาการไปถึงทางเลือก [60]

สเปกตรัมของแนวปฏิบัติจากกรณีศึกษาในบทนี้แสดงเกณฑ์การใช้ได้ชัดเจน ปลายหนึ่งคือกรณีที่การแก้โดยสันนิษฐานหลีกเลี่ยงไม่ได้ เช่น กาตุลลุส เมื่อทุกสำเนาสืบจากต้นแบบเดียวที่เสียหาย ความเสียหายจึงถูกถ่ายทอดซ้ำทุกสาย และจุดที่ข้อความทำให้ฉันทลักษณ์แตกหรือให้ความหมายไร้สาระย่อมชี้ข้อผิดพลาดที่ตัวต้นแบบ ผู้ชำระตั้งแต่ยุคมนุษยนิยมจึงแก้แบบเป็นระบบ แม้แต่ซาลูตาติเจ้าของตัวเขียน R เองก็เติมการแก้โดยสันนิษฐานของตนลงในเล่มตั้งแต่ปลายศตวรรษที่ 14 อีกฟากหนึ่งคือกรณีที่การแก้ต้องถูกระงับ เช่น เบโอวุลฟ์ เมื่อเทคโนโลยีภาพฟื้นตัวอักษรที่เคยคิดว่าอ่านไม่ได้ได้กว่า 100 จุด การอ่านที่เคยเป็นข้อสันนิษฐานจึงถูกแทนที่ด้วยการอ่านที่มาจากพยานจริง ฉบับชำระรุ่นที่สี่จึงใช้ภาคนิทานแบบเบา ผูกติดกับตัวอักษรที่ถูกอ่านคืนจากภาพ มิใช่เติมข้อความตามจินตนาการ หลักการคั่นกลางคือเมื่อเทคโนโลยีอ่านพยานได้เพิ่มขึ้น เขตอำนาจของการสันนิษฐานต้องหดลงเท่ากัน และทุกการแก้ที่ยังจำเป็นต้องถูกติดธงให้ผู้อ่านแยกได้ทันทีระหว่างสิ่งที่พยานมีอยู่จริงกับสิ่งที่ผู้ชำระเสนอเพิ่ม หลักการนี้คือเส้นแบ่งระหว่างการชำระข้อความเอกาตถีสมัยใหม่กับการแต่งข้อความขึ้นใหม่ตามรสนิยมซึ่งเป็นอดีตที่วงการเคยผ่านมาแล้ว

## สรุป

สถานการณ์ที่สามของต้นฉบับศึกษาคือสถานการณ์ที่ระเบียบวิธีมาตรฐานหมดอำนาจและต้องถูกแทนที่ทั้งระบบ เมื่อพยานเหลือเพียงฉบับเดียว แผนภาพสายสืบฉบับไม่มีวัตถุดิบให้สร้าง การสอบเทียบฉบับไม่มีคู่เทียบ และค่านิยามของการชำระข้อความเปลี่ยนจากการสืบสร้างไปสู่การทำสำเนาเทียบต้นอย่างซื่อสัตย์ กรณีศึกษาทั้งหมดในบทนี้ชี้บทเรียนร่วมห้าประการ ประการแรก ข้ออ้างความเป็นเอกาตถีเป็นข้ออ้างเชิงเงื่อนไขเสมอ กรณีตัวเขียนวินเชสเตอร์พิสูจน์ว่าพยานที่สองอาจปรากฏขึ้นเมื่อใดก็ได้และบีบให้ประวัติศาสตร์การชำระทั้งสายถูกเจรจาใหม่ ประการที่สอง ความเป็นเอกาตถีมีหลายชั้น ตั้งแต่พยานกายภาพเล่มเดียว พยานที่ถูกทำลายจนเหลือสำเนาแทน จนถึงเอกาตถีโดยตัวแทนของกาตุลลุส และการวินิจฉัยระเบียบวิธีต้องระบุชั้นให้ถูกก่อนเสมอ ประการที่สาม เมื่อไม่มีพยานคู่เทียบ การอ่านซ้ำด้วยเครื่องมือที่ดีกว่าทำหน้าที่แทนการสอบเทียบ ภาพหลายช่วงคลื่น แสงอัลตราไวโอเลต การเรืองแสงเอกซเรย์ และการหาอายุคาร์บอนกัมมันตรังสีที่บูรณาการกับการวิเคราะห์รูปร่างอักษรและการตัดสินของผู้เชี่ยวชาญโดยระบุความไม่แน่นอนอย่างชัดเจน ได้กลายเป็นเครื่องมือชำระข้อความในความหมายเต็มตัว ประการที่สี่ การอนุรักษ์สร้างและทำลายหลักฐานไปพร้อมกัน กรอบอนุรักษ์ปี 1845 ของเบโอวุลฟ์เป็นคำเตือนถาวรว่าการช่วยวัตถุอาจซ่อนข้อความ และสำเนาดิจิทัลเปิดคือหน้าที่ของการเอาใจใส่ทั้งต่อวัตถุและต่อสาธารณะ ประการที่ห้า ปมจารึกหลักที่ 1 สอนว่าความจริงแท้ของเอกาตถีต้องถูกตรวจสอบโดยแยกคำถามทางวัสดุออกจากคำถามทางอักษรศาสตร์ และโดยการเทียบเคียงกับคลังจารึกรวมเมื่อพยานคู่เทียบไม่มีอยู่

ทั้งหมดนี้ชี้ว่าในสถานการณ์เอกาตถี ศูนย์ถ่วงของการชำระข้อความเลื่อนจากความสัมพันธ์ระหว่างฉบับ ไปสู่ความสัมพันธ์ระหว่างพยานเดียวกับเครื่องมือ ทั้งเครื่องมือภาพ เครื่องมือเคมี วิธีสถิติ และวินัยการระบุแหล่งที่มาของทุกการอ่าน บทถัดไปจะขยายมุมมองนี้สู่ประเพณีตะวันออกและไทย ทั้งสี่วิธีชำระของเฉินหยวน วิชาการรวบรวมข้อความที่สูญจากคัมภีร์รวมจีน วิชาอักษรศาสตร์ไทย และการชำระพระไตรปิฎกรัชกาลที่ 5 เพื่อทดสอบว่ากรอบวิธีวิทยาสามสถานการณ์ที่สังเคราะห์มาจนถึงจุดนี้ ยืนหยัดได้เพียงใดเมื่อวางเคียงกับระเบียบวิธีที่พัฒนาในวัฒนธรรมต้นฉบับอื่น
