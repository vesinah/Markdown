# 4.4 คู่มือการสกัดและแปลงฐานข้อมูลเหมรังษีเป็น PAGE XML

คู่มือนี้อธิบายโครงสร้างฐานข้อมูล SQLite ของโปรแกรม **เหมรังษี (Hemrangsri)** และการใช้สคริปต์ Python ในการแปลงพิกัดและกล่องข้อความอ้างอิงให้เป็นมาตรฐาน **PAGE XML** ซึ่งเป็นรูปแบบอ้างอิงสำหรับการจัดเตรียมข้อมูลภาพและคำถอดอักษรเพื่อใช้เทรนโมเดล HTR ระดับบรรทัด (Line-based) เช่น **Kraken** หรือ **eScriptorium**

---

## 1. ความเข้าใจโครงสร้างฐานข้อมูลเหมรังษี (.hemr)

โปรเจกต์ของเหมรังษีจะถูกบันทึกเป็นไฟล์นามสกุล `.hemr` ซึ่งภายในเป็นฐานข้อมูล **SQLite** โครงสร้างที่เกี่ยวข้องโดยตรงกับการสกัดพื้นที่ข้อความ (Regions), บรรทัด (Lines) และข้อมูลการถอดอักษร (Transcriptions) มีดังนี้:

### 1.1 ตารางหลักที่ใช้ในการสกัดข้อมูล

1. **`images`**: เก็บสารระบบข้อมูลรูปภาพทั้งหมดในโปรเจกต์
   - `id`: รหัสประจำตัวรูปภาพ (UUID)
   - `name`: ชื่อของรูปภาพ
   - `originalPath`: พาธไฟล์รูปภาพต้นฉบับบนคอมพิวเตอร์
   - `width` / `height`: ขนาดกว้างและสูงของภาพพิกเซล
2. **`annotations`**: เก็บพิกัดและกล่องข้อความอ้างอิงที่ผู้ใช้ลากไว้บนภาพ
   - `id`: รหัสประจำตัวกล่อง
   - `imageId`: รหัสรูปภาพที่กล่องนี้สังกัดอยู่
   - `type`: ประเภทของกล่อง เช่น:
     - `'ประโยค/ข้อความ'` (Sentence/TextLine): เส้นบรรทัดสำหรับแปลผลระดับข้อความ
     - `'ตัวอักษร'` (Character/Glyph): พิกัดและคำอ่านอักษรเดี่ยว
     - `'ภาพ'` (Image/Region): พื้นที่ภาพวาดหรือสัญลักษณ์
   - `x`, `y`, `width`, `height`: พิกัดสี่เหลี่ยม Bounding Box
   - `points`: พิกัดรูปแบบ Polygon พจนานุกรมพิกัดแบบ JSON เช่น `[{"x":100,"y":120}, ...]` สำหรับรองรับขอบเขตโค้งงอตามรูปแบบใบลาน
   - `text`: ข้อความถอดอักษร (Transcription)
3. **`bannwiphat_items`**: เก็บข้อมูลเชิงโครงสร้างและลำดับชั้น (Hierarchy) ของใบลานและสมุดไทย
   - `id`: รหัสประจำรายการ
   - `type`: ประเภทของความสัมพันธ์เชิงโครงสร้าง เช่น `'หน้าเอกสาร'` (Page/Folio)
   - `rectX`, `rectY`, `rectW`, `rectH`: พิกัดของหน้านั้น ๆ บนภาพหลัก
   - `parentId`: รหัสอ้างอิงของโฟลเดอร์หรือระดับชั้นที่เหนือกว่า
   - `childrenIds`: รายการรหัสลูกย่อยในระดับถัดไป (เช่น เส้นบรรทัดที่เป็นลูกของหน้า) บันทึกเป็นอาเรย์ JSON
   - `value`: ป้ายกำกับชื่อหน้าหรือเลขใบลาน เช่น `'11v กํ'`, `'12v กะ'`

---

## 2. โครงสร้างและมาตรฐานของ PAGE XML

PAGE XML เป็นรูปแบบเอกสาร XML ที่เป็นมาตรฐานสากลในวงการแปลงภาพเอกสาร (Document Image Analysis) โครงสร้างหลักที่สคริปต์นี้สร้างประกอบด้วย:

```xml
<?xml version="1.0" encoding="UTF-8"?>
<PcGts xmlns="http://schema.primaresearch.org/PAGE/gts/pagecontent/2019-07-15" 
       xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance" 
       xsi:schemaLocation="http://schema.primaresearch.org/PAGE/gts/pagecontent/2019-07-15 http://schema.primaresearch.org/PAGE/gts/pagecontent/2019-07-15/pagecontent.xsd">
  <Metadata>
    <Creator>Hemrangsri PAGE XML Exporter</Creator>
    <Created>2026-06-04T23:00:00Z</Created>
  </Metadata>
  <Page imageFilename="image_001.jpg" imageWidth="8000" imageHeight="1000">
    <TextRegion id="region_0">
      <Coords points="0,0 8000,0 8000,1000 0,1000"/>
      <TextLine id="line_1780221529291">
        <Coords points="4022,1075 4446,1075 4446,1206 4022,1206"/>
        <TextEquiv>
          <Unicode>กมลิํ คจฺฉติ</Unicode>
        </TextEquiv>
      </TextLine>
    </TextRegion>
  </Page>
</PcGts>
```

## 3. สคริปต์สกัดข้อความเหมรังษีเป็น PAGE XML (`export_hemr_to_pagexml.py`)

คุณสามารถคัดลอกสคริปต์ Python ด้านล่างนี้ไปบันทึกเป็นไฟล์ `export_hemr_to_pagexml.py` สคริปต์นี้พัฒนาขึ้นโดยใช้โมดูลมาตรฐานของ Python ทั้งหมด (Standard Libraries) ทำให้ผู้เรียนรู้จากศูนย์สามารถเริ่มใช้งานได้ทันทีโดยไม่ต้องเผชิญปัญหาติดตั้งไลบรารีภายนอก

### โค้ดสคริปต์ฉบับสมบูรณ์

```python
# -*- coding: utf-8 -*-
"""
สคริปต์สกัดฐานข้อมูลเหมรังษี (.hemr) เป็นชุดข้อมูลภาพและ PAGE XML สำหรับเทรน Kraken
"""

import os
import sqlite3
import json
import shutil
import argparse
import xml.etree.ElementTree as ET
from xml.dom import minidom
from datetime import datetime, timezone

def parse_args():
    parser = argparse.ArgumentParser(description="สกัดข้อมูลเหมรังษี (.hemr) เป็น PAGE XML และรูปภาพคู่กัน")
    parser.add_argument("-i", "--input-db", required=True, help="พาธไปยังไฟล์ฐานข้อมูล .hemr (SQLite)")
    parser.add_argument("-o", "--output-dir", required=True, help="พาธโฟลเดอร์สำหรับบันทึกผลลัพธ์ (Output Directory)")
    parser.add_argument("-g", "--export-glyphs", action="store_true", default=False, 
                        help="เปิดใช้งานการสกัดข้อมูลอักษรเดี่ยว (Glyphs) ลงในไฟล์ XML ด้วย (ปิดโดยค่าเริ่มต้น)")
    return parser.parse_args()

def get_polygon_points_str(points_json, x, y, w, h):
    """
    แปลงค่าพิกัดรูปแบบพอยต์ JSON หรือค่า Bounding Box ให้เป็นสตริงพิกัดมาตรฐานของ PAGE XML
    """
    if points_json:
        try:
            pts = json.loads(points_json)
            if isinstance(pts, list) and len(pts) > 0:
                return " ".join(f"{int(pt['x'])},{int(pt['y'])}" for pt in pts)
        except Exception:
            pass
    # กรณีไม่มีพิกัดโค้งแบบ Polygon ให้ใช้ Bounding Box สี่เหลี่ยมแทน
    x1, y1 = int(x), int(y)
    x2, y2 = x1 + int(w), y1 + int(h)
    return f"{x1},{y1} {x2},{y1} {x2},{y2} {x1},{y2}"

def point_in_rect(px, py, rx, ry, rw, rh):
    """
    ตรวจสอบเชิงเรขาคณิตว่าพิกัดจุด (px, py) อยู่ภายในพื้นที่สี่เหลี่ยมของ Region (rx, ry, rw, rh) หรือไม่
    """
    return rx <= px <= (rx + rw) and ry <= py <= (ry + rh)

def export_project(db_path, output_dir, export_glyphs=False):
    print(f"[*] เริ่มต้นประมวลผลฐานข้อมูล: {db_path}")
    if not os.path.exists(db_path):
        print(f"[!] ไม่พบไฟล์ฐานข้อมูลในพาธที่ระบุ: {db_path}")
        return False

    os.makedirs(output_dir, exist_ok=True)
    
    # เชื่อมต่อ SQLite
    conn = sqlite3.connect(db_path)
    # อนุญาตให้เข้าถึงคอลัมน์ผ่านชื่อคีย์
    conn.row_factory = sqlite3.Row
    cur = conn.cursor()

    try:
        # 1. ดึงรายการรูปภาพ
        cur.execute("SELECT id, name, originalPath, width, height FROM images")
        images = cur.fetchall()
        print(f"[*] ตรวจพบรูปภาพในระบบทั้งหมด: {len(images)} รูปภาพ")

        for img in images:
            img_id = img["id"]
            img_name = img["name"]
            original_path = img["originalPath"]
            img_width = img["width"] or 0
            img_height = img["height"] or 0

            # กำหนดชื่อไฟล์รูปภาพปลายทางและพาธ XML
            img_filename = os.path.basename(original_path) if original_path else f"{img_name}.jpg"
            img_base_name, _ = os.path.splitext(img_filename)
            xml_filename = f"{img_base_name}.xml"
            xml_path = os.path.join(output_dir, xml_filename)
            dest_img_path = os.path.join(output_dir, img_filename)

            # คัดลอกภาพต้นฉบับคู่กัน (ถ้าพบไฟล์)
            copied = False
            if original_path and os.path.exists(original_path):
                try:
                    shutil.copy2(original_path, dest_img_path)
                    copied = True
                except Exception as e:
                    print(f"  [!] ไม่สามารถคัดลอกภาพ {original_path}: {e}")
            else:
                # ลองค้นหาในโฟลเดอร์เดียวกับฐานข้อมูล
                local_search = os.path.join(os.path.dirname(db_path), img_filename)
                if os.path.exists(local_search):
                    try:
                        shutil.copy2(local_search, dest_img_path)
                        copied = True
                    except Exception:
                        pass

            if not copied:
                print(f"  [!] คำเตือน: ไม่พบไฟล์ภาพจริงของ {img_name} บนดิสก์ (สร้างเฉพาะไฟล์ XML)")

            # 2. ตรวจสอบคอลัมน์ในตาราง bannwiphat_items และ annotations ป้องกันความผิดพลาดหากไม่มีคอลัมน์ type หรือ parentId
            cur.execute("PRAGMA table_info(bannwiphat_items)")
            b_cols = [col[1] for col in cur.fetchall()]
            has_b_type = "type" in b_cols
            has_b_parentId = "parentId" in b_cols

            b_query = "SELECT id, rectX, rectY, rectW, rectH, value FROM bannwiphat_items WHERE imageId = ?"
            if has_b_type and has_b_parentId:
                b_query += " AND (type = 'หน้าเอกสาร' OR parentId IS NULL)"
            elif has_b_type:
                b_query += " AND type = 'หน้าเอกสาร'"
            elif has_b_parentId:
                b_query += " AND parentId IS NULL"

            cur.execute(b_query, (img_id,))
            db_regions = cur.fetchall()

            cur.execute("PRAGMA table_info(annotations)")
            ann_cols = [col[1] for col in cur.fetchall()]
            has_ann_type = "type" in ann_cols

            if has_ann_type:
                cur.execute("""
                    SELECT id, x, y, width, height, text 
                    FROM annotations 
                    WHERE imageId = ? AND (type = 'ภาพ' OR type = 'IMAGE' OR type = 'region')
                """, (img_id,))
            else:
                cur.execute("""
                    SELECT id, x, y, width, height, text 
                    FROM annotations 
                    WHERE imageId = ?
                """, (img_id,))
            db_ann_regions = cur.fetchall()

            regions = []
            # ประกอบโครงสร้าง Region
            for r in db_regions:
                regions.append({
                    "id": f"region_b_{r['id']}",
                    "x": r["rectX"] or 0,
                    "y": r["rectY"] or 0,
                    "w": r["rectW"] or img_width,
                    "h": r["rectH"] or img_height,
                    "label": r["value"] or "Folio"
                })
            for r in db_ann_regions:
                regions.append({
                    "id": f"region_a_{r['id']}",
                    "x": r["x"] or 0,
                    "y": r["y"] or 0,
                    "w": r["width"] or img_width,
                    "h": r["height"] or img_height,
                    "label": r["text"] or "AnnotationRegion"
                })

            # หากไม่พบกรอบพื้นที่ใดเลย ให้สร้าง Default Region ครอบคลุมทั้งหน้า
            if len(regions) == 0:
                regions.append({
                    "id": "region_default",
                    "x": 0,
                    "y": 0,
                    "w": img_width or 8000,
                    "h": img_height or 1000,
                    "label": "DefaultContentRegion"
                })

            # 3. ดึงเส้นบรรทัดข้อความ (Annotations)
            if has_ann_type:
                cur.execute("""
                    SELECT id, type, x, y, width, height, points, text, transcription 
                    FROM annotations 
                    WHERE imageId = ?
                """, (img_id,))
            else:
                cur.execute("""
                    SELECT id, x, y, width, height, points, text, transcription 
                    FROM annotations 
                    WHERE imageId = ?
                """, (img_id,))
            all_annotations = cur.fetchall()
            
            # กรองเฉพาะบรรทัดข้อความ
            lines_data = []
            glyphs_data = []

            for ann in all_annotations:
                type_str = str(ann["type"]) if has_ann_type and ann["type"] else ""
                
                # หากมีคอลัมน์ type ให้เช็คประเภท หากไม่มีคอลัมน์ type ให้เหมาว่าทั้งหมดเป็นบรรทัดข้อความ
                is_line = True
                is_glyph = False
                if has_ann_type:
                    if type_str in ("ประโยค/ข้อความ", "SENTENCE", "TextLine", "line"):
                        is_line = True
                    elif type_str in ("ตัวอักษร", "GLYPH", "character"):
                        is_line = False
                        is_glyph = True
                    else:
                        is_line = False
                
                # รองรับคีย์ text หรือ transcription
                text_content = ""
                if "text" in ann.keys() and ann["text"]:
                    text_content = ann["text"]
                elif "transcription" in ann.keys() and ann["transcription"]:
                    text_content = ann["transcription"]

                if is_line:
                    lines_data.append({
                        "id": ann["id"],
                        "x": ann["x"] or 0,
                        "y": ann["y"] or 0,
                        "w": ann["width"] or 0,
                        "h": ann["height"] or 0,
                        "points": ann["points"] if "points" in ann.keys() else None,
                        "text": text_content
                    })
                elif is_glyph:
                    glyphs_data.append({
                        "id": ann["id"],
                        "x": ann["x"] or 0,
                        "y": ann["y"] or 0,
                        "w": ann["width"] or 0,
                        "h": ann["height"] or 0,
                        "points": ann["points"] if "points" in ann.keys() else None,
                        "char": text_content
                    })

            # 4. แปลงโครงสร้างบรรทัดและอักษรเดี่ยวลง Region
            region_lines = {r["id"]: [] for r in regions}
            
            # ฟังก์ชันคำนวณหา Region ที่บรรทัดสังกัดอยู่ (ตรวจจับจากกึ่งกลางจุด)
            for line in lines_data:
                center_x = line["x"] + (line["w"] / 2)
                center_y = line["y"] + (line["h"] / 2)
                
                assigned = False
                for r in regions:
                    # ข้าม default region ไว้ตรวจสอบท้ายสุดเพื่อความแม่นยำ
                    if r["id"] == "region_default":
                        continue
                    if point_in_rect(center_x, center_y, r["x"], r["y"], r["w"], r["h"]):
                        region_lines[r["id"]].append(line)
                        assigned = True
                        break
                
                if not assigned:
                    # ถ้าไม่มี Region ไหนครอบคลุมเลย หรือไม่มีพิกัด ให้ลงที่ default
                    default_id = "region_default" if "region_default" in region_lines else regions[0]["id"]
                    region_lines[default_id].append(line)

            # ประมวลผล Glyphs (ถ้าเปิดใช้งาน)
            line_glyphs = {line["id"]: [] for line in lines_data}
            if export_glyphs and len(glyphs_data) > 0:
                for glyph in glyphs_data:
                    g_center_x = glyph["x"] + (glyph["w"] / 2)
                    g_center_y = glyph["y"] + (glyph["h"] / 2)
                    
                    # ค้นหาเส้นบรรทัดที่ตัวอักษรนี้ซ้อนอยู่
                    for line in lines_data:
                        if point_in_rect(g_center_x, g_center_y, line["x"], line["y"], line["w"], line["h"]):
                            line_glyphs[line["id"]].append(glyph)
                            break

            # 5. สร้างโครงสร้าง PAGE XML ด้วย ElementTree
            # ประกาศ Namespace มาตรฐาน และลงทะเบียนเพื่อป้องกันพรีฟิกซ์ ns0
            ns_uri = "http://schema.primaresearch.org/PAGE/gts/pagecontent/2019-07-15"
            ET.register_namespace("", ns_uri)
            
            def ns_tag(tag):
                return f"{{{ns_uri}}}{tag}"
            
            root = ET.Element(ns_tag("PcGts"), {
                "xmlns:xsi": "http://www.w3.org/2001/XMLSchema-instance",
                "xsi:schemaLocation": f"{ns_uri} {ns_uri}/pagecontent.xsd"
            })
            
            # ส่วน Metadata
            meta = ET.SubElement(root, ns_tag("Metadata"))
            creator = ET.SubElement(meta, ns_tag("Creator"))
            creator.text = "Hemrangsri PAGE XML Exporter"
            created = ET.SubElement(meta, ns_tag("Created"))
            created.text = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")

            # ส่วนหน้ากระดาษ (Page)
            page = ET.SubElement(root, ns_tag("Page"), {
                "imageFilename": img_filename,
                "imageWidth": str(int(img_width)),
                "imageHeight": str(int(img_height))
            })

            # วาดทุก Region และข้าม Region ที่ว่างเปล่า (ยกเว้น default region เพื่อความปลอดภัย)
            for r in regions:
                lines_in_r = region_lines[r["id"]]
                if len(lines_in_r) == 0 and r["id"] != "region_default":
                    continue # ไม่วาดบล็อกว่างเพื่อความสะอาดของข้อมูล
                
                text_region = ET.SubElement(page, ns_tag("TextRegion"), {
                    "id": r["id"],
                    "type": "paragraph"
                })
                
                # พิกัด Region
                r_pts = f"{int(r['x'])},{int(r['y'])} {int(r['x'])+int(r['w'])},{int(r['y'])} {int(r['x'])+int(r['w'])},{int(r['y'])+int(r['h'])} {int(r['x'])},{int(r['y'])+int(r['h'])}"
                ET.SubElement(text_region, ns_tag("Coords"), {"points": r_pts})
                
                # วาดบรรทัด (TextLine) ภายใต้ Region
                for line in lines_in_r:
                    text_line = ET.SubElement(text_region, ns_tag("TextLine"), {
                        "id": f"line_{line['id']}"
                    })
                    
                    # แปลงพิกัดของบรรทัด
                    line_pts = get_polygon_points_str(line["points"], line["x"], line["y"], line["w"], line["h"])
                    ET.SubElement(text_line, ns_tag("Coords"), {"points": line_pts})
                    
                    # แปลงข้อมูลตัวอักษรเดี่ยว (ถ้ามี)
                    if export_glyphs and line["id"] in line_glyphs:
                        for glyph in line_glyphs[line["id"]]:
                            glyph_el = ET.SubElement(text_line, ns_tag("Glyph"), {
                                "id": f"glyph_{glyph['id']}"
                            })
                            g_pts = get_polygon_points_str(glyph["points"], glyph["x"], glyph["y"], glyph["w"], glyph["h"])
                            ET.SubElement(glyph_el, ns_tag("Coords"), {"points": g_pts})
                            
                            g_equiv = ET.SubElement(glyph_el, ns_tag("TextEquiv"))
                            g_uni = ET.SubElement(g_equiv, ns_tag("Unicode"))
                            g_uni.text = glyph["char"]

                    # คำถอดอักษรของบรรทัด
                    text_equiv = ET.SubElement(text_line, ns_tag("TextEquiv"))
                    unicode_el = ET.SubElement(text_equiv, ns_tag("Unicode"))
                    unicode_el.text = line["text"]

            # บันทึกไฟล์ PAGE XML พร้อมทำ Pretty Print ให้อ่านง่าย
            xml_str = ET.tostring(root, encoding="utf-8")
            parsed_xml = minidom.parseString(xml_str)
            pretty_xml = parsed_xml.toprettyxml(indent="  ", encoding="utf-8")
            
            with open(xml_path, "wb") as f:
                f.write(pretty_xml)
            
            print(f"  [+] สำเร็จ: เขียนไฟล์ '{xml_filename}' (ดึงได้ {len(lines_data)} บรรทัด)")

    except Exception as e:
        print(f"[!] เกิดข้อผิดพลาดในขณะทำงาน: {e}")
        import traceback
        traceback.print_exc()
        return False
    finally:
        conn.close()
    
    print("[*] การส่งออกเสร็จสมบูรณ์เรียบร้อยแล้ว!")
    return True

if __name__ == "__main__":
    args = parse_args()
    export_project(args.input_db, args.output_dir, args.export_glyphs)
```
